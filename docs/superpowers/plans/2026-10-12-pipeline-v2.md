```python
"""OpenAI cagirislari: chat_json (gpt-4o-mini), generate_image / edit_image (gpt-image-2). Acar .env-den."""
from __future__ import annotations

import base64
import json
import os
import time
import urllib.error
import urllib.request
import uuid

BASE = "https://api.openai.com/v1"
TEXT_MODEL = "gpt-4o-mini"
IMAGE_MODEL = "gpt-image-2"
RETRIES = 6
TIMEOUT_S = 300
ENV = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), ".env")


class LLMError(Exception):
    pass


def api_key() -> str:
    key = os.environ.get("OPENAI_API_KEY", "")
    if not key and os.path.isfile(ENV):
        for line in open(ENV, encoding="utf-8"):
            if line.strip().startswith("OPENAI_API_KEY="):
                key = line.split("=", 1)[1].strip().strip('"')
    if not key:
        raise LLMError("OPENAI_API_KEY yoxdur (.env)")
    return key


def send(url: str, body: bytes, content_type: str, *, opener=urllib.request.urlopen, sleep=time.sleep) -> dict:
    headers = {"Content-Type": content_type, "Authorization": f"Bearer {api_key()}"}
    last = ""
    for attempt in range(RETRIES):
        req = urllib.request.Request(url, data=body, headers=headers, method="POST")
        try:
            with opener(req, timeout=TIMEOUT_S) as r:
                return json.loads(r.read())
        except urllib.error.HTTPError as e:
            last = f"HTTP {e.code}: {e.read().decode('utf-8', 'replace')[:300]}"
            if "insufficient_quota" in last:
                raise LLMError("OpenAI balansi bitib: " + last) from e
            if e.code not in (408, 409, 429) and e.code < 500:
                raise LLMError(last) from e
        except (urllib.error.URLError, TimeoutError, json.JSONDecodeError) as e:
            last = f"{type(e).__name__}: {e}"
        sleep(min(2 ** attempt, 30))
    raise LLMError(f"{RETRIES} cehdden sonra ugursuz - {last}")


def chat_json(system: str, user: str, *, model: str = TEXT_MODEL, max_tokens: int = 4000) -> dict:
    payload = {"model": model, "temperature": 0.4, "max_tokens": max_tokens,
               "response_format": {"type": "json_object"},
               "messages": [{"role": "system", "content": system}, {"role": "user", "content": user}]}
    res = send(f"{BASE}/chat/completions", json.dumps(payload).encode("utf-8"), "application/json")
    try:
        return json.loads(res["choices"][0]["message"]["content"])
    except (KeyError, IndexError, json.JSONDecodeError) as e:
        raise LLMError("gozlenilmeyen chat cavabi") from e


def _image(res: dict) -> bytes:
    try:
        return base64.b64decode(res["data"][0]["b64_json"])
    except (KeyError, IndexError, TypeError) as e:
        raise LLMError("gozlenilmeyen sekil cavabi") from e


def generate_image(prompt: str, *, size: str = "1536x1024", quality: str = "medium") -> bytes:
    payload = {"model": IMAGE_MODEL, "prompt": prompt, "size": size, "quality": quality, "n": 1}
    return _image(send(f"{BASE}/images/generations", json.dumps(payload).encode("utf-8"), "application/json"))


def _multipart(fields: dict[str, str], files: dict[str, str]) -> tuple[bytes, str]:
    b = uuid.uuid4().hex
    out = bytearray()
    for k, v in fields.items():
        out += f'--{b}\r\nContent-Disposition: form-data; name="{k}"\r\n\r\n{v}\r\n'.encode("utf-8")
    for k, path in files.items():
        out += (f'--{b}\r\nContent-Disposition: form-data; name="{k}"; filename="{os.path.basename(path)}"\r\n'
                "Content-Type: image/png\r\n\r\n").encode("utf-8")
        out += open(path, "rb").read() + b"\r\n"
    out += f"--{b}--\r\n".encode("utf-8")
    return bytes(out), f"multipart/form-data; boundary={b}"


def edit_image(prompt: str, ref_path: str, *, size: str = "1024x1536", quality: str = "low") -> bytes:
    body, ctype = _multipart({"model": IMAGE_MODEL, "prompt": prompt, "size": size, "quality": quality,
                              "background": "opaque", "n": "1"}, {"image[]": ref_path})
    return _image(send(f"{BASE}/images/edits", body, ctype))
```

```python
"""Ssenari .md -> basliq + bolmeler -> sehneler. Ssenari YOXLANMIR, metn sozbesoz qalir (yalniz ** * --- silinir).
Iki format: (A) **NARRATION:** / **VISUAL:** isareli istehsal ssenarisi; (B) sade metn (her abzas danisiqdir)."""
from __future__ import annotations

import os
import re

TIMECODE = re.compile(r"^\s*\d{1,2}:\d{2}\s*[–-]\s*\d{1,2}:\d{2}\s*[—–-]\s*")
LABEL = re.compile(r"^\*\*\s*([A-Za-z ]+?)\s*:\s*\*\*\s*(.*)$")     # **VISUAL:** text
SENTENCE = re.compile(r"(?:(?<=[.!?…])|(?<=[.!?…][\"”’)]))\s+(?=[\"“‘(]?[A-Z0-9$])")


def clean(text: str) -> str:
    text = text.replace("**", "").replace("__", "")
    text = re.sub(r"(?<!\w)\*(?!\s)(.+?)(?<!\s)\*(?!\w)", r"\1", text)
    return " ".join(text.split())


def _blocks(lines: list[str]) -> list[str]:
    out, cur = [], []
    for ln in lines + [""]:
        if ln.strip() and ln.strip() != "---":
            cur.append(ln.strip())
        elif cur:
            out.append(" ".join(cur))
            cur = []
    return out


def parse(markdown: str, fallback_title: str) -> dict:
    lines = markdown.replace("\r\n", "\n").split("\n")
    title, heads = "", []                      # heads: [(name, [lines])]
    pre: list[str] = []
    for ln in lines:
        if ln.startswith("# ") and not title:
            title = clean(ln[2:])
        elif ln.startswith("## "):
            heads.append((clean(TIMECODE.sub("", ln[3:])), []))
        elif heads:
            heads[-1][1].append(ln)
        else:
            pre.append(ln)
    marked = "NARRATION:" in markdown.upper()
    sections = []
    if not marked and _blocks(pre):
        sections.append({"name": "", "visual": "", "paragraphs": [clean(b) for b in _blocks(pre)]})
    for name, body in heads:
        visual, paras, mode = "", [], "narr" if not marked else "skip"
        for b in _blocks(body):
            m = LABEL.match(b)
            if m:
                label, rest = m.group(1).upper(), m.group(2)
                if label == "VISUAL":
                    visual, mode = clean(rest), "skip"
                elif label == "NARRATION":
                    mode = "narr"
                    if rest.strip():
                        paras.append(clean(rest))
                else:
                    mode = "skip" if marked else mode      # **Format:** kimi meta setir
                continue
            if mode == "narr" and not b.startswith(("- ", "|", "* ")):
                paras.append(clean(b))
        if paras:
            sections.append({"name": name, "visual": visual, "paragraphs": paras})
    return {"title": title or fallback_title, "sections": sections}


def _sentences(paragraph: str) -> list[str]:
    return [s.strip() for s in SENTENCE.split(paragraph) if s.strip()]


def split_scenes(sections: list[dict], max_words: int = 40) -> list[dict]:
    scenes: list[dict] = []
    for sec in sections:
        cur: list[str] = []
        first = True

        def flush() -> None:
            nonlocal cur, first
            if cur:
                scenes.append({"n": len(scenes) + 1, "section": sec["name"], "section_start": first,
                               "visual": sec["visual"], "text": " ".join(cur), "type": None,
                               "photo_prompt": "", "owl": "", "anim": None, "duration": 0.0})
                cur, first = [], False
        for para in sec["paragraphs"]:
            for s in _sentences(para):
                if cur and len(" ".join(cur + [s]).split()) > max_words:
                    flush()
                cur.append(s)
        flush()
    return scenes


def slugify(title: str) -> str:
    return re.sub(r"[^a-z0-9]+", "-", title.lower().replace("’", "").replace("'", "")).strip("-")[:80]


def build(path: str) -> dict:
    with open(path, encoding="utf-8") as f:
        parsed = parse(f.read(), os.path.splitext(os.path.basename(path))[0])
```

```python
"""Ekrandaki animasiya YALNIZ sehne danisigindan: her reqem danisiqda yazilan ve ya sozle deyilen deyerdir, her yazi
parcasinin sozleri danisiqda var. Uygun gelmese None -> plan quote (danisiqdan cumle) gosterir. Ssenari yoxlanmir."""
from __future__ import annotations

import re

KINDS = ("counter", "bars", "percent", "flow", "list", "versus", "timeline", "quote")
SCALES = {"thousand": 1e3, "million": 1e6, "billion": 1e9, "trillion": 1e12}
UNITS = {w: i for i, w in enumerate("zero one two three four five six seven eight nine ten eleven twelve thirteen "
                                    "fourteen fifteen sixteen seventeen eighteen nineteen".split())}
TENS = {w: 10 * (i + 2) for i, w in enumerate("twenty thirty forty fifty sixty seventy eighty ninety".split())}
DIGITS = re.compile(r"(\d[\d,]*(?:\.\d+)?)\s*(thousand|million|billion|trillion)?", re.I)
WORD = re.compile(r"[a-z]+(?:-[a-z]+)?", re.I)
MAX_ITEMS = 4


def _spoken(text: str) -> set[float]:
    out: set[float] = set()
    total = cur = 0.0
    seen = False
    for w in [t.lower() for t in WORD.findall(text)] + ["."]:
        parts = w.split("-")
        if all(p in UNITS or p in TENS for p in parts):
            cur += sum(UNITS.get(p, TENS.get(p, 0)) for p in parts)
            seen = True
        elif w == "hundred" and seen:
            cur = (cur or 1) * 100
        elif w in SCALES and seen:
            total += (cur or 1) * SCALES[w]
            cur = 0.0
        elif w == "a" and not seen:
            continue
        else:
            if seen:
                out.add(total + cur)
            total = cur = 0.0
            seen = False
    return out


def numbers_in(text: str) -> set[float]:
    out = set()
    for m in DIGITS.finditer(text):
        v = float(m.group(1).replace(",", ""))
        out.add(v)
        if m.group(2):
            out.add(v * SCALES[m.group(2).lower()])
    return out | _spoken(text)


def _stem(w: str) -> str:
    return w.lower().rstrip("s")


def words_ok(label: str, text: str) -> bool:
    have = {_stem(w) for w in re.findall(r"[A-Za-z]+", text)}
    words = [w for w in re.findall(r"[A-Za-z]+", label) if len(w) >= 3]
    return bool(label.strip()) and all(_stem(w) in have for w in words)


def _num(x) -> float | None:
    try:
        return float(str(x).replace(",", "").replace("$", "").replace("%", ""))
    except (TypeError, ValueError):
        return None


def _num_ok(x, nums: set[float]) -> float | None:
    v = _num(x)
    return v if v is not None and any(abs(v - n) <= 1e-6 * max(1.0, abs(n)) for n in nums) else None


def _strs(xs, text: str) -> list[str] | None:
    if not isinstance(xs, list):
        return None
    xs = [str(x).strip() for x in xs][:MAX_ITEMS]
    return xs if len(xs) >= 2 and all(words_ok(x, text) for x in xs) else None


def validate(spec: dict, text: str) -> dict | None:
    if not isinstance(spec, dict) or spec.get("kind") not in KINDS:
        return None
    kind, nums = spec["kind"], numbers_in(text)
    if kind == "counter":
        v = _num_ok(spec.get("value"), nums)
        label = str(spec.get("label", "")).strip()
        suffix = spec.get("suffix", "") if spec.get("suffix", "") in ("", "%", " thousand", " million", " billion") else ""
        prefix = "$" if spec.get("prefix") == "$" else ""
        return ({"kind": kind, "value": v, "prefix": prefix, "suffix": suffix, "label": label}
                if v is not None and words_ok(label, text) else None)
    if kind == "percent":
        v = _num_ok(spec.get("value"), nums)
        label = str(spec.get("label", "")).strip()
        return {"kind": kind, "value": v, "label": label} if v is not None and 0 <= v <= 100 and words_ok(label, text) else None
    if kind == "bars":
        items = spec.get("items") if isinstance(spec.get("items"), list) else []
        good = [{"label": str(i.get("label", "")).strip(), "value": _num_ok(i.get("value"), nums)}
                for i in items[:MAX_ITEMS] if isinstance(i, dict)]
        good = [i for i in good if i["value"] is not None and words_ok(i["label"], text)]
        return {"kind": kind, "items": good} if len(good) >= 2 and len(good) == len(items[:MAX_ITEMS]) else None
    if kind in ("flow", "list", "timeline"):
        key = {"flow": "steps", "list": "items", "timeline": "events"}[kind]
        xs = _strs(spec.get(key), text)
        return {"kind": kind, key: xs} if xs else None
    if kind == "versus":
        left, right = str(spec.get("left", "")).strip(), str(spec.get("right", "")).strip()
        return {"kind": kind, "left": left, "right": right} if words_ok(left, text) and words_ok(right, text) else None
    q = str(spec.get("text", "")).strip()
    return {"kind": "quote", "text": q} if q and q.rstrip("…").strip() in text else None


def quote_for(text: str) -> dict:
    sentences = [s.strip() for s in re.split(r"(?<=[.!?])\s+", text) if s.strip()]
    short = [s for s in sentences if len(s.split()) <= 16]
    if short:
        return {"kind": "quote", "text": max(short, key=len)}
    return {"kind": "quote", "text": " ".join(text.split()[:14]) + "…"}
```

```python
"""Plan: sehnelerin 40%-i animasiya (beraber paylanmis), 60%-i foto. GPT her sehneye foto promptu, bayqus pozasi ve
animasiya teklifleri verir; animasiya yalniz danisiqdan (grounding), nov novbelesir. GPT xetasi -> VISUAL + quote."""
from __future__ import annotations

import json
import math

import grounding
import llm

ANIM_SHARE = 0.40
KIND_CAP = 0.25
ROTATION = [k for k in grounding.KINDS if k != "quote"]

SYSTEM = """You plan visuals for an explainer video. For EVERY scene return:
- "photo_prompt": one realistic photograph (12-30 words) that shows what the narration and the section's VISUAL
  direction describe. Realistic documentary photo, natural light. No readable text, letters or logos in the photo.
- "owl": what the cartoon owl presenter does in this scene (6-14 words): pose, expression and at most one prop that
  fits the narration.
- "anim": a list of 3 animation ideas of DIFFERENT kinds, best first, for this scene. Use ONLY numbers and words
  that appear in the scene narration - never compute, round or invent. Kinds and fields:
  counter {value, prefix "$" or "", suffix "", "%", " thousand", " million" or " billion", label}
  bars {items: [{label, value}] 2-4}
  percent {value 0-100, label}
  flow {steps: [2-4 short phrases]}
  list {items: [2-4 short phrases]}
  versus {left, right}
  timeline {events: [2-4 short phrases]}
  Labels and phrases are 1-5 words copied from the narration.
Answer ONLY JSON: {"scenes": [{"n", "photo_prompt", "owl", "anim": [...]}]}"""


def assign_types(n: int) -> list[str]:
    return ["anim" if math.floor((i + 1) * ANIM_SHARE + 0.5) > math.floor(i * ANIM_SHARE + 0.5) else "photo"
            for i in range(n)]


def kind_order(i: int) -> list[str]:
    return ROTATION[i % len(ROTATION):] + ROTATION[:i % len(ROTATION)]


def pick_anim(candidates: list[dict], text: str, prev_kind: str | None, counts: dict[str, int], n_anim: int,
              variant: int) -> dict:
    cap = max(1, math.floor(n_anim * KIND_CAP))
    for c in candidates or []:
        spec = grounding.validate(c, text)
        if spec and spec["kind"] != "quote" and spec["kind"] != prev_kind and counts.get(spec["kind"], 0) < cap:
            return {**spec, "variant": variant}
    return {**grounding.quote_for(text), "variant": variant}


def _user(scenes: list[dict], title: str) -> str:
    parts = [f"VIDEO: {title}"]
    for s in scenes:
        parts.append(f"SCENE {s['n']}: section={s['section']!r}\nVISUAL: {s['visual']}\nNARRATION: {s['text']}")
    return "\n\n".join(parts)


def _ask_batch(scenes: list[dict], title: str, ask) -> dict[int, dict]:
    try:
        res = ask(SYSTEM, _user(scenes, title))
        return {int(r["n"]): r for r in res.get("scenes", []) if isinstance(r, dict) and "n" in r}
    except Exception as e:  # noqa: BLE001 - plan hec vaxt dayanmir
        print(f"  plan: GPT xetasi ({str(e)[:120]}) - VISUAL + quote istifade olunur", flush=True)
        return {}


def plan_scenes(data: dict, ask=llm.chat_json, batch: int = 12) -> dict:
    scenes = [dict(s) for s in data["scenes"]]
    types = assign_types(len(scenes))
    got: dict[int, dict] = {}
    for i in range(0, len(scenes), batch):
        got.update(_ask_batch(scenes[i:i + batch], data["title"], ask))
    n_anim, counts, prev, k = types.count("anim"), {}, None, 0
    for s, t in zip(scenes, types):
        r = got.get(s["n"], {})
        s["type"] = t
        s["photo_prompt"] = str(r.get("photo_prompt") or s["visual"] or s["text"])[:400]
        s["owl"] = str(r.get("owl") or "standing and explaining, friendly")[:200]
        s["anim"] = None
        if t == "anim":
            cands = r.get("anim") if isinstance(r.get("anim"), list) else []
            order = kind_order(k)
            cands = sorted(cands, key=lambda c: order.index(c["kind"]) if isinstance(c, dict) and c.get("kind") in order else 99)
            s["anim"] = pick_anim(cands, s["text"], prev, counts, n_anim, k % 2)
            counts[s["anim"]["kind"]] = counts.get(s["anim"]["kind"], 0) + 1
            prev, k = s["anim"]["kind"], k + 1
    return {**data, "scenes": scenes}


def main(ep: str) -> None:
    path = f"{ep}/scenes.json"
    data = json.load(open(path, encoding="utf-8"))
    out = plan_scenes(data)
    json.dump(out, open(path, "w", encoding="utf-8"), indent=2, ensure_ascii=False)
    kinds = [s["anim"]["kind"] for s in out["scenes"] if s["anim"]]
    print(f"  plan: {len(out['scenes'])} sehne, animasiya {len(kinds)}: "
          + ", ".join(f"{k}x{kinds.count(k)}" for k in sorted(set(kinds))), flush=True)
```

```python
Alinmasa referans sprite istifade olunur - bayqus her kadrda ekrandadir."""
from __future__ import annotations

import io
import json
import os
import shutil
from concurrent.futures import ThreadPoolExecutor

import numpy as np
from PIL import Image

import llm

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
OWL_REF = os.path.join(ROOT, "Character", "ELI5_Owl", "sprites_hd", "front.png")
ATTEMPTS = 3
CHARACTER = ("The exact same cartoon owl character as in the reference image: same brown feathers and "
             "feather tuft, same round black glasses, same navy suit, white shirt, yellow tie and brown belt, "
             "same proportions, face and 3D Pixar-like art style.")
RULES = ("Only this one owl (plus the single prop if mentioned). Full body from head to feet, feet visible, "
         "standing, centered. Background: a perfectly flat, uniform, solid pure magenta (#FF00FF) colour "
         "filling the whole image - no gradient, no floor, no shadow, no texture, no checkerboard. "
         "Nothing magenta or pink on the owl or the prop. No text, no letters, no people, no other animals.")
KEY_BORDER, KEY_SPREAD_MAX, KEY_DIST, KEY_SOLID, KEY_MIN_ALPHA = 4, 0.05, 40.0, 0.12, 0.06


def owl_prompt(action: str) -> str:
    return f"{CHARACTER} The owl is {action}. {RULES}"


def key_out(im: Image.Image) -> Image.Image:
    c = np.asarray(im.convert("RGB")).astype(np.float64)
    b = KEY_BORDER
    border = np.concatenate([c[:b].reshape(-1, 3), c[-b:].reshape(-1, 3), c[:, :b].reshape(-1, 3), c[:, -b:].reshape(-1, 3)])
    k = np.median(border, axis=0)
    if not (k[0] > 180 and k[2] > 180 and k[1] < 90):
        raise ValueError("fon magenta deyil")
    if float(np.mean(np.linalg.norm(border - k, axis=1) > KEY_DIST)) > KEY_SPREAD_MAX:
        raise ValueError("fon bircins deyil")
    km = min(k[0], k[2]) - k[1]
    m = np.minimum(c[..., 0], c[..., 2]) - c[..., 1]
    a = np.where(m <= KEY_SOLID * km, 1.0, np.clip(1.0 - m / km, 0.0, 1.0))
    a[a < KEY_MIN_ALPHA] = 0.0
    safe = np.where(a > 0, a, 1.0)[..., None]
    fg = np.clip((c - (1.0 - a)[..., None] * k) / safe, 0, 255)
    return Image.fromarray(np.dstack([fg, a * 255.0]).round().astype(np.uint8), "RGBA")


def _crop(im: Image.Image) -> Image.Image:
    box = im.getchannel("A").point(lambda v: 255 if v > 16 else 0).getbbox()
    return im.crop(box) if box else im


def _one(path: str, action: str, edit) -> None:
    for i in range(ATTEMPTS):
        try:
            _crop(key_out(Image.open(io.BytesIO(edit(owl_prompt(action), OWL_REF))))).save(path)
            return
        except Exception as e:  # noqa: BLE001
            print(f"  bayqus {os.path.basename(path)} cehd {i + 1}: {str(e)[:120]}", flush=True)
    shutil.copyfile(OWL_REF, path)
    print(f"  bayqus {os.path.basename(path)}: referans sprite", flush=True)


def render_owls(ep: str, edit=lambda p, r: llm.edit_image(p, r), workers: int = 3) -> None:
    data = json.load(open(os.path.join(ep, "scenes.json"), encoding="utf-8"))
    os.makedirs(os.path.join(ep, "owl"), exist_ok=True)
    todo = [s for s in data["scenes"] if not os.path.isfile(os.path.join(ep, "owl", f"sc{s['n']:02d}.png"))]
    with ThreadPoolExecutor(workers) as pool:
        list(pool.map(lambda s: _one(os.path.join(ep, "owl", f"sc{s['n']:02d}.png"), s["owl"], edit), todo))
    print(f"  bayqus: {len(todo)} sehne", flush=True)
```

```python
"""Yalniz TTS ucun telaffuz: reqemler ingilis sozlerine. Ekran/altyazi metni deyismir."""
from __future__ import annotations

import re

from num2words import num2words

MONEY = re.compile(r"\$(\d[\d,]*(?:\.\d+)?)(\s*(?:thousand|million|billion|trillion))?")
PERCENT = re.compile(r"(\d[\d,]*(?:\.\d+)?)\s?%")
NUMBER = re.compile(r"\d[\d,]*(?:\.\d+)?")


def _words(raw: str) -> str:
    v = float(raw.replace(",", ""))
    return num2words(int(v) if v.is_integer() else v)


def _number(m: re.Match) -> str:
    raw = m.group(0)
    if re.fullmatch(r"(19|20)\d\d", raw):
        return num2words(int(raw), to="year")
    return _words(raw)


def to_speech(text: str) -> str:
    text = MONEY.sub(lambda m: f"{_words(m.group(1))}{m.group(2) or ''} dollars", text)
    text = PERCENT.sub(lambda m: f"{_words(m.group(1))} percent", text)
    return NUMBER.sub(_number, text)
```

```python
"""Altyazi ssenari metninden (sozbesoz); vaxt sehne sesi icinde herf uzunluguna gore paylanir."""
from __future__ import annotations

import json
import os
import re


def chunks(text: str, max_words: int = 7) -> list[str]:
    out, cur = [], []
    for w in text.split():
        cur.append(w)
        if len(cur) >= max_words or (len(cur) >= 3 and re.search(r"[,.;:!?]$", w)):
            out.append(" ".join(cur))
            cur = []
    if cur:
        out.append(" ".join(cur))
    return out


def build(scenes: list[dict], tail: float = 0.35) -> list[dict]:
    caps, t0 = [], 0.0
    for s in scenes:
        speak = max(0.1, s["duration"] - tail)
        parts = chunks(s["text"])
        total = sum(len(p) for p in parts) or 1
        t = t0
        for p in parts:
            d = speak * len(p) / total
            caps.append({"start": round(t, 3), "end": round(t + d, 3), "text": p})
            t += d
        caps[-1]["end"] = round(t0 + s["duration"], 3)
        t0 += s["duration"]
    return caps


def _ts(x: float) -> str:
    ms = int(round(x * 1000))
    return f"{ms // 3600000:02d}:{ms // 60000 % 60:02d}:{ms // 1000 % 60:02d},{ms % 1000:03d}"


def to_srt(caps: list[dict]) -> str:
    return "\n".join(f"{i}\n{_ts(c['start'])} --> {_ts(c['end'])}\n{c['text']}\n" for i, c in enumerate(caps, 1))


def main(ep: str) -> None:
    data = json.load(open(os.path.join(ep, "scenes.json"), encoding="utf-8"))
    caps = build(data["scenes"])
    json.dump(caps, open(os.path.join(ep, "captions.json"), "w", encoding="utf-8"), indent=1, ensure_ascii=False)
    open(os.path.join(ep, f"{data['slug']}.srt"), "w", encoding="utf-8").write(to_srt(caps))
```

```python
"""Stable Audio Open (MusicGen\\.venv, lokal GPU): video uzunlugunda fon musiqisi -> music.wav.
Klipler crossfade ile birlesir; video daha uzundursa mix merhelesi dovr etdirir.
Istifade: MusicGen\\.venv\\Scripts\\python Projects\\music.py Episodes\\<slug>"""
from __future__ import annotations

import hashlib
import json
import math
import os
import subprocess
import sys

MODEL = "stabilityai/stable-audio-open-1.0"
MODEL_FILES = ["model_index.json", "projection_model/*", "scheduler/*", "text_encoder/*", "tokenizer/*",
               "transformer/*", "vae/*"]
CLIP_S, FADE_S, STEPS, MAX_CLIPS = 45.0, 3.0, 100, 8
NEGATIVE = "vocals, singing, speech, voice, low quality, distortion, noise, harsh, loud drums"
STYLES = (
    "Light upbeat corporate background music, soft acoustic guitar, warm piano, positive, 110 BPM, instrumental",
    "Calm lo-fi explainer background, mellow electric piano, soft bass, light brushed drums, 90 BPM, instrumental",
    "Modern minimal documentary background, soft piano, warm cello, subtle pulse, 100 BPM, instrumental",
    "Sleek corporate ambient, deep synth bass, soft electric piano chords, steady light beat, 105 BPM, instrumental",
)


def clips_needed(total_s: float) -> int:
    return max(1, min(MAX_CLIPS, math.ceil(total_s / (CLIP_S - FADE_S))))


def main(ep: str) -> None:
    os.environ.setdefault("HF_HUB_DISABLE_XET", "1")
    import soundfile as sf
    import torch
    from diffusers import StableAudioPipeline
    from huggingface_hub import snapshot_download

    data = json.load(open(os.path.join(ep, "scenes.json"), encoding="utf-8"))
    total = sum(s["duration"] for s in data["scenes"])
    seed = int.from_bytes(hashlib.sha1(data["slug"].encode()).digest()[:4], "big")
    style = STYLES[seed % len(STYLES)]
    pipe = StableAudioPipeline.from_pretrained(snapshot_download(MODEL, allow_patterns=MODEL_FILES),
                                               torch_dtype=torch.float16).to("cuda")
    parts_dir = os.path.join(ep, "music_parts")
    os.makedirs(parts_dir, exist_ok=True)
    paths = []
    for i in range(clips_needed(total)):
        p = os.path.join(parts_dir, f"part{i + 1:02d}.wav")
        if not os.path.isfile(p):
            audio = pipe(style, negative_prompt=NEGATIVE, num_inference_steps=STEPS, audio_end_in_s=CLIP_S,
                         generator=torch.Generator("cuda").manual_seed(seed + i)).audios[0]
            sf.write(p, audio.T.float().cpu().numpy(), pipe.vae.sampling_rate)
        paths.append(p)
        print(f"  musiqi {i + 1}/{clips_needed(total)}", flush=True)
    out = os.path.join(ep, "music.wav")
    if len(paths) == 1:
        subprocess.run(["ffmpeg", "-y", "-loglevel", "error", "-i", paths[0], out], check=True)
        return
    inputs = sum((["-i", p] for p in paths), [])
    chain, last = [], "[0:a]"
    for i in range(1, len(paths)):
        tag = f"[a{i}]"
        chain.append(f"{last}[{i}:a]acrossfade=d={FADE_S}:c1=tri:c2=tri{tag}")
        last = tag
    subprocess.run(["ffmpeg", "-y", "-loglevel", "error", *inputs, "-filter_complex", ";".join(chain),
                    "-map", last, out], check=True)


if __name__ == "__main__":
    main(sys.argv[1])
```

```python
"""Remotion render: public qovlugu (foto JPEG, bayqus PNG, sriftler) + props.json -> video_raw.mp4 (sessiz)."""
from __future__ import annotations

import json
import os
import shutil
import subprocess

from PIL import Image

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
REMOTION = os.path.join(ROOT, "Remotion")
FONTS = os.path.join(REMOTION, "public-static", "fonts")


def props(data: dict, caps: list[dict], fps: int = 30) -> dict:
    scenes, at = [], 0
    for s in data["scenes"]:
        frames = max(1, round(s["duration"] * fps))
        scenes.append({"n": s["n"], "type": s["type"], "img": f"img/sc{s['n']:02d}.jpg" if s["type"] == "photo" else None,
                       "owl": f"owl/sc{s['n']:02d}.png", "anim": s["anim"], "section": s["section"],
                       "sectionStart": bool(s["section_start"]), "from": at, "frames": frames})
        at += frames
    return {"fps": fps, "total": at, "scenes": scenes,
            "captions": [{"from": round(c["start"] * fps), "to": round(c["end"] * fps), "text": c["text"]} for c in caps]}


def public_dir(ep: str, data: dict) -> str:
    pub = os.path.join(ep, "public")
    for sub in ("img", "owl"):
        os.makedirs(os.path.join(pub, sub), exist_ok=True)
    shutil.copytree(FONTS, os.path.join(pub, "fonts"), dirs_exist_ok=True)
    for s in data["scenes"]:
        n = f"sc{s['n']:02d}"
        if s["type"] == "photo":
            with Image.open(os.path.join(ep, "img", f"{n}.png")) as im:
                im.convert("RGB").resize((1920, 1280), Image.LANCZOS).save(os.path.join(pub, "img", f"{n}.jpg"), quality=92)
        shutil.copyfile(os.path.join(ep, "owl", f"{n}.png"), os.path.join(pub, "owl", f"{n}.png"))
    return pub


def main(ep: str, runner=subprocess.run) -> None:
    ep = os.path.abspath(ep)
    data = json.load(open(os.path.join(ep, "scenes.json"), encoding="utf-8"))
    caps = json.load(open(os.path.join(ep, "captions.json"), encoding="utf-8"))
    pub = public_dir(ep, data)
    pp = os.path.join(ep, "props.json")
    json.dump(props(data, caps), open(pp, "w", encoding="utf-8"), ensure_ascii=False)
    npx = "npx.cmd" if os.name == "nt" else "npx"
    runner([npx, "remotion", "render", "src/index.ts", "Episode", os.path.join(ep, "video_raw.mp4"),
            f"--props={pp}", f"--public-dir={pub}", "--muted", "--codec=h264", "--crf=18", "--log=info"],
           cwd=REMOTION, check=True)
```

```python
import shutil

import llm

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))

def _mmss(s: float) -> str:
    s = int(s)
    return f"{s // 3600}:{s // 60 % 60:02d}:{s % 60:02d}" if s >= 3600 else f"{s // 60}:{s % 60:02d}"


def chapters(scenes: list[dict]) -> list[str]:
    out, t = [], 0.0
    for s in scenes:
        if s["section_start"] and s["section"]:
            out.append(f"{_mmss(t)} {s['section']}")
        t += s["duration"]
    return out


def youtube_text(data: dict) -> str:
    tags = ", ".join(w.lower() for w in re.findall(r"[A-Za-z]+", data["title"]) if len(w) > 2)
    desc = data["scenes"][0]["text"] if data["scenes"] else ""
    return "\n".join([data["title"], "", desc, "", *chapters(data["scenes"]), "", f"Tags: {tags}"]) + "\n"


def _wrap(draw, text, font, width):
    lines, cur = [], ""
    for w in text.split():
        test = f"{cur} {w}".strip()
        if draw.textlength(test, font=font) <= width:
            cur = test
        else:
            lines.append(cur)
            cur = w
    return lines + [cur]


def thumbnail(ep: str, data: dict, gen) -> Image.Image:
    first = next((s for s in data["scenes"] if s["type"] == "photo"), data["scenes"][0])
    try:
        bg = Image.open(io.BytesIO(gen(first["photo_prompt"] + " Realistic photo, no text.")))
    except Exception as e:  # noqa: BLE001
        print(f"  thumbnail fonu alinmadi ({str(e)[:100]}) - ilk foto", flush=True)
        p = os.path.join(ep, "img", f"sc{first['n']:02d}.png")
        bg = Image.open(p) if os.path.isfile(p) else Image.new("RGB", (1536, 1024), (11, 19, 36))
    bg = bg.convert("RGB").resize((1280, 720))
    owl = Image.open(os.path.join(ep, "owl", "sc01.png")).convert("RGBA")
    owl = owl.resize((int(owl.width * 600 / owl.height), 600))
    bg.paste(owl, (1280 - owl.width - 20, 720 - owl.height), owl)
    d = ImageDraw.Draw(bg)
    font = ImageFont.truetype(FONT, 74)
    y = 60
    for line in _wrap(d, data["title"].upper(), font, 760):
        d.text((50, y), line, font=font, fill="white", stroke_width=6, stroke_fill="black")
```

```python
"""Merheleler ardicil; her merhele fayl yazir, fayl varsa kecilir (resume). Xeta -> 1 tekrar; sehne seviyyesinde
fallback-ler modullarin icindedir, ona gore is ssenari sebebile hec vaxt dayanmir."""
from __future__ import annotations

import json
import os
import subprocess
import sys
import time
from dataclasses import dataclass
from typing import Callable

HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.dirname(HERE)
sys.path.insert(0, HERE)
import captions  # noqa: E402
import images  # noqa: E402
import mix  # noqa: E402
import owl  # noqa: E402
import plan  # noqa: E402
import publish  # noqa: E402
import render  # noqa: E402
import script  # noqa: E402

TTS_PY = os.path.join(ROOT, "TTS", ".venv", "Scripts", "python.exe")
MUSIC_PY = os.path.join(ROOT, "MusicGen", ".venv", "Scripts", "python.exe")


@dataclass(frozen=True)
class Stage:
    name: str
    done: Callable[[str], bool]
    run: Callable[[str], None]


def _f(*parts: str) -> Callable[[str], bool]:
    return lambda ep: os.path.isfile(os.path.join(ep, *parts))


def _scenes(ep: str) -> dict:
    return json.load(open(os.path.join(ep, "scenes.json"), encoding="utf-8"))


def _all(sub: str, ext: str, photo_only: bool = False) -> Callable[[str], bool]:
    def done(ep: str) -> bool:
        if not os.path.isfile(os.path.join(ep, "scenes.json")):
            return False
        return all(os.path.isfile(os.path.join(ep, sub, f"sc{s['n']:02d}.{ext}")) for s in _scenes(ep)["scenes"]
                   if not photo_only or s["type"] == "photo")
    return done


def _planned(ep: str) -> bool:
    return os.path.isfile(os.path.join(ep, "scenes.json")) and all(s["type"] for s in _scenes(ep)["scenes"])


def _voice(ep: str) -> None:
    subprocess.run([TTS_PY, os.path.join(HERE, "voice.py"), ep], check=True)


def _music(ep: str) -> None:
    try:
        subprocess.run([MUSIC_PY, os.path.join(HERE, "music.py"), ep], check=True)
    except Exception as e:  # noqa: BLE001 - musiqisiz davam
        print(f"  musiqi alinmadi ({e}) - musiqisiz davam", flush=True)
        open(os.path.join(ep, "music.skip"), "w").close()


def _slug_mp4(ep: str) -> bool:
    return os.path.isfile(os.path.join(ep, "scenes.json")) and os.path.isfile(os.path.join(ep, f"{_scenes(ep)['slug']}.mp4"))


def _published(ep: str) -> bool:
    return os.path.isfile(os.path.join(ep, "scenes.json")) and os.path.isfile(
        os.path.join(ROOT, "Hazir_Videolar", _scenes(ep)["slug"], "youtube.txt"))


STAGES = [
    Stage("script", _f("scenes.json"), lambda ep: None),          # start() yazir
    Stage("plan", _planned, plan.main),
    Stage("images", _all("img", "png", photo_only=True), images.render_photos),
    Stage("owl", _all("owl", "png"), owl.render_owls),
    Stage("voice", _f("narration.wav"), _voice),
    Stage("captions", _f("captions.json"), captions.main),
    Stage("music", lambda ep: _f("music.wav")(ep) or _f("music.skip")(ep), _music),
    Stage("render", _f("video_raw.mp4"), render.main),
    Stage("mix", _slug_mp4, mix.main),
    Stage("publish", _published, publish.main),
]


def run_stages(ep: str, stages=STAGES, log=print) -> None:
    for i, st in enumerate(stages, 1):
        if st.done(ep):
            log(f"[{i}/{len(stages)}] {st.name}: hazirdir - kecilir")
            continue
        log(f"[{i}/{len(stages)}] {st.name} ... ({time.strftime('%H:%M:%S')})")
        for attempt in (1, 2):
            try:
                st.run(ep)
                break
            except Exception as e:  # noqa: BLE001
                log(f"  {st.name} xetasi (cehd {attempt}): {str(e)[:300]}")
                if attempt == 2:
                    log(f"DAYANDI: {st.name}. Davam: python run.py --resume {os.path.basename(ep)}")
                    raise SystemExit(1)


def find_script(name: str) -> str:
    if os.path.isfile(name):
        return os.path.abspath(name)
    base = os.path.join(ROOT, "Ssenariler", os.path.basename(name))
    for p in (base, base + ".md"):
        if os.path.isfile(p):
            return p
    raise SystemExit(f"ssenari tapilmadi: {name}")


def start(script_path: str) -> str:
    data = script.build(find_script(script_path))
    ep = os.path.join(ROOT, "Episodes", data["slug"])
    os.makedirs(ep, exist_ok=True)
    if not os.path.isfile(os.path.join(ep, "scenes.json")):
        json.dump(data, open(os.path.join(ep, "scenes.json"), "w", encoding="utf-8"), indent=2, ensure_ascii=False)
    return ep


def main(argv: list[str]) -> None:
    ep = os.path.join(ROOT, "Episodes", argv[1]) if argv[:1] == ["--resume"] else start(argv[0])
    log_path = os.path.join(ep, "run.log")

    def log(msg: str) -> None:
        print(msg, flush=True)
        with open(log_path, "a", encoding="utf-8") as f:
            f.write(msg + "\n")
    log(f"=== {os.path.basename(ep)} ===")
    run_stages(ep, STAGES, log)
    log("BITDI")
```
